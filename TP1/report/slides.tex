                  \small 83\% reduction in disk reads
            \vspace{\baselineskip}
            \normalsize
            \item \texttt{SinglePass} $\rightarrow$ \texttt{FastLoad}: \\
            \begin{center}

                \vspace{\baselineskip}

                2 stages, 25 tasks
                \vspace{\baselineskip}
            \end{center}
            \item \texttt{DuckDB} and \texttt{PostgreSQL}: \\
                  \small Different frameworks
        \end{itemize}
    \end{minipage}
